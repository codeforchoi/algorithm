import java.io.*;
import java.util.StringTokenizer;

/**
 * 페르마의 소정리
 * a^(p-1) == 1
 * a^(p-2) == 1/a
 * nCr == n! / (n-r)! * r! == n! * 1/(n-r)! * 1/r!
 * nCr % p = n! % p * ((n-r)!)^(p-2) % p * ((r!)^(p-2)) % p
 * 나눗셈의 나머지 정리는 없고 곱셉의 나머지 정리 사용
 * 분할정복 알고리즘을 사용
 */
public class Solution {
	
	private static final long P = 1_234_567_891;
	private static long[] memoFactorial;
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		StringTokenizer st;
		
		int T = Integer.parseInt(br.readLine());
		
		for(int tc = 1; tc <= T; tc++) {
			st = new StringTokenizer(br.readLine());
			
			int N = Integer.parseInt(st.nextToken());
			int R = Integer.parseInt(st.nextToken());
			
			factorial(1_000_000);
			
			long numerator = memoFactorial[N];
			long denominator_nr = memoFactorial[N - R];
			long denominator_r = memoFactorial[R];
			
			denominator_nr = divideAndConquer(denominator_nr, P - 2);
			denominator_r = divideAndConquer(denominator_r, P - 2);
			
			long answer = ((numerator * denominator_nr) % P * denominator_r) % P;
			
			sb.append("#").append(tc).append(" ").append(answer).append("\n");
		}
		System.out.println(sb);
	}
	
	private static void factorial(int n) {
		memoFactorial = new long[n + 1];
		
		memoFactorial[0] = 1;
		
		for(int i = 1; i <= n; i++) {
			memoFactorial[i] = i * memoFactorial[i - 1] % P;
		}
	}
	
	private static long divideAndConquer(long n, long p) {
		if(p == 0) return 1;
		
		long result = divideAndConquer(n, p / 2);
		result = result * result % P;
		
		return (p % 2 == 0) ? result : result * n % P; 		
	}	
}
